cmake_minimum_required(VERSION 3.16)
project(TextResolution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(text_resolution
    src/main.cpp src/tokenizer.cpp src/nlp_parser.cpp src/logic.cpp src/resolution.cpp src/other_methods.cpp)
target_include_directories(text_resolution PRIVATE include)
if(MSVC)
    target_compile_options(text_resolution PRIVATE /W4)
else()
    target_compile_options(text_resolution PRIVATE -Wall -Wextra -Wpedantic)
endif()
